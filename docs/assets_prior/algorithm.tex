\begin{algorithm}[t]
\caption{Generation of one proximal dataset.}
\label{alg:prior}
\begin{algorithmic}[1]
\Require sequence length $T$, Monte Carlo count $K$, hyperparameters.
\State sample $d_U$ from \eqref{eq:dU_triangular}; sample $d_W,d_Z,d_X$ from \eqref{eq:proxy_dims}
\State instantiate random MLP mechanisms $f_X,f_U,f_W,f_Z,f_A,f_Y,f_{\mathrm{aug}}$
\State sample per-dataset hyperparameters: noise shares, confounding shares $s_A,s_Y$, edge keep-probabilities, overlap knobs $s,b$, $\gamma\sim\mathrm{Unif}(0,1)$
\Statex\vspace{2pt}
\Repeat
  \State $X \gets \mathrm{Corrupt}(f_X(\varepsilon_X)) \in \R^{T\times d_X}$, tiled $K$ times and standardised to $\tilde X$
  \State subsample covariate inputs $\tilde X_U,\tilde X_W,\tilde X_Z,\tilde X_A,\tilde X_Y\subseteq\tilde X$
  \State draw independent noise $\varepsilon_U,\varepsilon_W,\varepsilon_Z,\varepsilon_A,\varepsilon_Y$ over all $TK$ rows
  \State $U \gets \mathrm{Corrupt}(f_U(\tilde X_U,\varepsilon_U))$, standardised to $\tilde U$
  \State $W \gets \mathrm{Corrupt}(f_W(\tilde U,\tilde X_W,\varepsilon_W))$, standardised to $\tilde W$ \Comment{$Z\not\to f_W$}
  \State $Z \gets \mathrm{Corrupt}(f_Z(\tilde U,\tilde X_Z,\varepsilon_Z))$, standardised to $\tilde Z$ \Comment{$W\not\to f_Z$}
  \State subsample direct-proxy inputs $\tilde Z_A\subseteq\tilde Z$ and $\tilde W_Y\subseteq\tilde W$
  \State form $U^{[A]},U^{[Y]}$ from $\tilde U$ and $U^{\mathrm{aug}}=f_{\mathrm{aug}}(\tilde U)$ via confounding shares $s_A,s_Y$
  \State $\ell \gets f_A(U^{[A]},\tilde X_A,\tilde Z_A,\varepsilon_A)$;\quad $\tilde\ell \gets (\ell-\bar\ell)/(\operatorname{sd}(\ell)+10^{-8})$
  \State $\pi \gets \operatorname{clip}\big(\sigma(s\,\tilde\ell+b),\,\epsilon,\,1-\epsilon\big)$
  \State $A \sim \mathrm{Bernoulli}(\pi)$
  \State $(Y_0^{\mathrm{raw}},Y_1^{\mathrm{raw}}) \gets f_Y(U^{[Y]},\tilde X_Y,\tilde W_Y,\varepsilon_Y)$
  \State draw $\alpha\sim\mathrm{Unif}(0,1)^{TK}$; remix to $(Y_0,Y_1)$ via \eqref{eq:remix}
  \State $\hat\mu_a(x_i) \gets \tfrac1K\sum_{k=1}^K y_{i,k}(a)$ for $a\in\{0,1\}$
  \State select $k_i\sim\mathrm{Unif}\{1,\dots,K\}$ per unit; keep observed $\{x_i,w_i,z_i,a_i,y_i\}$
  \State remove constant columns from $X,U,W,Z$; update active dims $d_X,d_U,d_W,d_Z$ \label{line:constant_column_filter}
\Until{$d_X,d_W,d_Z>0$,  $\min(d_W,d_Z)\ge d_U>0$ and categorical dominance of $W,Z$ over $U$.}
\Statex\vspace{2pt}
\State \Return observed $\{x_i,w_i,z_i,a_i,y_i\}_{i=1}^T$ and targets $\{\hat\mu_0(x_i),\hat\mu_1(x_i)\}_{i=1}^T$
\end{algorithmic}
\end{algorithm}
